                             <p class="mb-4 text-gray-700">Ahora, a dibujar. Recuerda, cada elemento en su propia capa nombrada correctamente.</p>

                             <ul class="space-y-3">

                                 <li><strong>Cabeza y Ojos:</strong> Dibuja la forma de la cabeza en su capa. Luego, crea un grupo para cada ojo (`+Ojo izquierdo`, `+Ojo derecho`). Dentro de cada uno, dibuja la `Pupila` y el `Párpado` en capas separadas.</li>

                                 <li><strong>Boca (Visemas):</strong> Este es el paso más detallado para una sincronización de labios (lip-sync) perfecta. Crea un grupo llamado `Boca` y dentro, dibuja cada forma en su propia capa.</li>

<script>

    document.addEventListener('DOMContentLoaded', function () {

        const tabs = document.querySelectorAll('.tab-btn');

        const sections = document.querySelectorAll('.content-section');

        const accordions = document.querySelectorAll('.accordion-item');


        tabs.forEach(tab => {

            tab.addEventListener('click', () => {

                const target = tab.getAttribute('data-target');


                tabs.forEach(t => {

                    t.classList.remove('tab-active');

                    t.classList.add('tab-inactive');

                });

                tab.classList.add('tab-active');

                tab.classList.remove('tab-inactive');


                sections.forEach(section => {

                    if (section.id === target) {

                        section.classList.remove('hidden');

                    } else {

                        section.classList.add('hidden');

                    }

                });

            });

        });


        accordions.forEach(item => {

            const header = item.querySelector('.accordion-header');

            const content = item.querySelector('.accordion-content');

            const icon = header.querySelector('span');


            header.addEventListener('click', () => {

                const isOpen = content.style.maxHeight;


                // Close all other accordions

                accordions.forEach(otherItem => {

                    if (otherItem !== item) {

                        otherItem.querySelector('.accordion-content').style.maxHeight = null;

                        otherItem.querySelector('.accordion-content').style.paddingTop = null;

                        otherItem.querySelector('.accordion-content').style.paddingBottom = null;

                        otherItem.querySelector('.accordion-header span').style.transform = 'rotate(0deg)';

                    }

                });

                

                if (isOpen) {

                    content.style.maxHeight = null;

                    content.style.paddingTop = null;

                    content.style.paddingBottom = null;

                    icon.style.transform = 'rotate(0deg)';

                } else {

                    content.style.maxHeight = content.scrollHeight + 40 + "px";

                    content.style.paddingTop = '1.25rem';

                    content.style.paddingBottom = '1.25rem';

                    icon.style.transform = 'rotate(180deg)';

                }

            });

        });

    });

</script>
